// Test-only codec composition: не OS process attribution и не activation proof.
package router_test

import (
 "encoding/json"
 "os"
 "path/filepath"
 "testing"
 R "github.com/xtls/xray-core/app/router"
 N "github.com/xtls/xray-core/common/net"
 "github.com/xtls/xray-core/features/routing"
 S "github.com/xtls/xray-core/features/routing/session"
 "github.com/xtls/xray-core/infra/conf"
)

type codecContext struct { S.Context; domain, ip, app, inbound string }
func(c *codecContext) GetTargetDomain() string { return c.domain }
func(c *codecContext) GetTargetIPs() []N.IP { return []N.IP{N.ParseIP(c.ip)} }
func(c *codecContext) GetInboundTag() string { return c.inbound }
// Только test dependency вместо OS lookup; wire path проверяется перед заменой.
type codecApp struct{}
func(codecApp) Apply(c routing.Context) bool { return c.(*codecContext).app=="/opt/fixture/app-0" }
func TestP04AActualCodecAND(t *testing.T) {
 root:=os.Getenv("P04A_DATA"); raw,err:=os.ReadFile(filepath.Join(root,"config.json")); if err!=nil {t.Fatal(err)}
 var envelope struct { Routing conf.RouterConfig `json:"routing"` }; if err=json.Unmarshal(raw,&envelope);err!=nil {t.Fatal(err)}
 config,err:=envelope.Routing.Build(); if err!=nil {t.Fatal(err)}
 condition,err:=config.Rule[2].BuildCondition();if err!=nil {t.Fatal(err)}
 chain,ok:=condition.(*R.ConditionChan);if !ok {t.Fatal("не AND chain")}
 processes:=0
 for i,item:=range *chain { if process,ok:=item.(*R.ProcessNameMatcher);ok {
  if len(process.AbsPaths)!=1 || process.AbsPaths[0]!="/opt/fixture/app-0" {t.Fatal("потерян executable")}
  (*chain)[i]=codecApp{};processes++
 }}
 if processes!=1 {t.Fatal("process conjunct отсутствует")}
 cases:=[]struct{name,domain,ip,app,inbound string; want bool}{
  {"match","a.example.invalid","203.0.113.7","/opt/fixture/app-0","system-tun",true},
  {"wrongapp","a.example.invalid","203.0.113.7","/opt/fixture/other","system-tun",false},
  {"wrongname","miss.example.invalid","203.0.113.7","/opt/fixture/app-0","system-tun",false},
  {"wrongCIDR","a.example.invalid","203.0.114.7","/opt/fixture/app-0","system-tun",false},
  {"wronginbound","a.example.invalid","203.0.113.7","/opt/fixture/app-0","socks",false},
 }
 for _,c:=range cases { t.Run(c.name,func(t *testing.T){got:=chain.Apply(&codecContext{domain:c.domain,ip:c.ip,app:c.app,inbound:c.inbound});if got!=c.want {t.Fatalf("match=%v want %v",got,c.want)}}) }
}
